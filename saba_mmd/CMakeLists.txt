cmake_minimum_required(VERSION 3.16)
project(CuCaT_Saba LANGUAGES C CXX)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static dependencies" FORCE)

# CuCaT ships only the target extractor and OpenGL renderer.
set(SABA_INSTALL OFF)
set(SABA_ENABLE_TEST OFF)
set(SABA_USE_GLSLANG OFF)
set(SABA_FORCE_GLFW_BUILD ON)
find_package(OpenGL REQUIRED)
add_subdirectory(external/bullet)
set(BULLET_INCLUDE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/external/bullet/src")
set(BULLET_LIBRARIES BulletDynamics BulletCollision LinearMath)
include_directories(src viewer ${BULLET_INCLUDE_DIRS})
foreach(dep gl3w gli glm stb spdlog tinyobjloader imgui json sol2 tinyxfileloader tinyddsloader)
    include_directories("${CMAKE_CURRENT_SOURCE_DIR}/external/${dep}/include")
endforeach()
include_directories(BEFORE "${CMAKE_CURRENT_SOURCE_DIR}/external/lua")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/external/glfw/include")
if(WIN32)
    add_compile_definitions(UNICODE _UNICODE)
endif()
if(MSVC)
    add_compile_options(/MP)
endif()
add_subdirectory(external)
add_subdirectory(src)
add_subdirectory(viewer)
add_executable(mmd_target_dump mmd_target_dump.cpp)
target_link_libraries(mmd_target_dump PRIVATE Saba)
add_executable(saba_viewer saba_viewer.cpp)
target_link_libraries(saba_viewer PRIVATE SabaViewer ${CMAKE_DL_LIBS})
# Keep generated runtime files beside the binaries for all generators.
foreach(app mmd_target_dump saba_viewer)
    set_target_properties(${app} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
endforeach()
add_custom_command(TARGET saba_viewer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/viewer/Saba/Viewer/resource"
    "$<TARGET_FILE_DIR:saba_viewer>/resource"
    VERBATIM)
